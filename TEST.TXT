generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")

enum Status {
  ACTIVE
  INACTIVE
  PENDING
  BLOCKED
}

enum UserAssignmentType {
  STATE
  DISTRICT
  BLOCK
  GP
  VILLAGE
  WARD
  BOOTH
}

enum ContactStatus {
  ACTIVE
  INACTIVE
  MOVED
  DECEASED
  DUPLICATE
}

enum ImportStatus {
  PENDING
  PROCESSING
  COMPLETED
  FAILED
}

model Role {
  id          String  @id @default(uuid())
  name        String  @unique
  slug        String  @unique
  description String?

  users           User[]
  rolePermissions RolePermission[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("roles")
}

model Menu {
  id String @id @default(uuid())

  name String
  slug String @unique

  path String?
  icon String?

  parentId String?

  parent   Menu?  @relation("MenuHierarchy", fields: [parentId], references: [id])
  children Menu[] @relation("MenuHierarchy")

  sortOrder Int @default(0)

  isActive Boolean @default(true)

  permissions RolePermission[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("menus")
}

model RolePermission {
  id String @id @default(uuid())

  roleId String
  menuId String

  role Role @relation(fields: [roleId], references: [id], onDelete: Cascade)
  menu Menu @relation(fields: [menuId], references: [id], onDelete: Cascade)

  canView   Boolean @default(false)
  canAdd    Boolean @default(false)
  canEdit   Boolean @default(false)
  canDelete Boolean @default(false)

  canImport Boolean @default(false)
  canExport Boolean @default(false)

  canDownload Boolean @default(false)
  canPrint    Boolean @default(false)

  canApprove Boolean @default(false)
  canReject  Boolean @default(false)

  createdAt DateTime @default(now())

  @@unique([roleId, menuId])
  @@index([roleId])
  @@index([menuId])
  @@map("role_permissions")
}

model User {
  id String @id @default(uuid())

  username String  @unique
  email    String? @unique
  mobile   String  @unique

  passwordHash String

  firstName String
  lastName  String?

  roleId String
  role   Role   @relation(fields: [roleId], references: [id])

  status Status @default(ACTIVE)

  isVerified Boolean @default(false)

  lastLoginAt DateTime?

  sessions   UserSession[]
  otpRecords OtpVerification[]

  assignments UserAssignment[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  deletedAt DateTime?

  @@index([roleId])
  @@index([status])
  @@map("users")
}

model UserAssignment {
  id String @id @default(uuid())

  userId String
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)

  type UserAssignmentType

  stateId    String?
  districtId String?
  blockId    String?
  gpId       String?
  villageId  String?
  wardId     String?
  boothId    String?

  state    State?    @relation(fields: [stateId], references: [id])
  district District? @relation(fields: [districtId], references: [id])
  block    Block?    @relation(fields: [blockId], references: [id])
  gp       GP?       @relation(fields: [gpId], references: [id])
  village  Village?  @relation(fields: [villageId], references: [id])
  ward     Ward?     @relation(fields: [wardId], references: [id])
  booth    Booth?    @relation(fields: [boothId], references: [id])

  createdAt DateTime @default(now())

  @@index([userId])
  @@map("user_assignments")
}

model UserSession {
  id String @id @default(uuid())

  userId String
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)

  jwtId String @unique

  ipAddress String?
  userAgent String?

  expiresAt DateTime

  revokedAt DateTime?

  createdAt DateTime @default(now())

  @@index([userId])

model OtpVerification {
  id String @id @default(uuid())

  userId String

model Contact {
  id String @id @default(uuid())

  fullName String

  fatherHusbandName String?

  gender String?

  email String?

  whatsapp String?

  address String?

  photoUrl String?

  occupationId String?
  occupation   Occupation? @relation(fields: [occupationId], references: [id])

  stateId    String
  districtId String
  blockId    String
  gpId       String
  villageId  String
  wardId     String
  boothId    String

  state    State    @relation(fields: [stateId], references: [id])
  district District @relation(fields: [districtId], references: [id])
  block    Block    @relation(fields: [blockId], references: [id])
  gp       GP       @relation(fields: [gpId], references: [id])
  village  Village  @relation(fields: [villageId], references: [id])
  ward     Ward     @relation(fields: [wardId], references: [id])
  booth    Booth    @relation(fields: [boothId], references: [id])

  phones ContactPhone[]

  remarks String?

  status ContactStatus @default(ACTIVE)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  deletedAt DateTime?

  @@index([fullName])
  @@index([occupationId])
  @@index([stateId])
  @@index([districtId])
  @@index([blockId])
  @@index([gpId])
  @@index([villageId])
  @@index([wardId])
  @@index([boothId])
  @@map("contacts")
}

model ContactPhone {

  phone String

  isPrimary Boolean @default(false)

  createdAt DateTime @default(now())

  @@index([contactId])
  @@index([phone])
  @@map("contact_phones")
}
